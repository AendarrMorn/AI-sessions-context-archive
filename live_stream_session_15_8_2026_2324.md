```cmd
# Пути к файлам на твоем сетевом диске (замени W:\ на свою букву)
$baseFile = "W:\cameras.txt"
$logFile  = "W:\status_log.csv"

# Логины и пароли от систем (впиши свои)
$trassirUser = "admin"; $trassirPass = "12345"
$rviUser     = "admin"; $rviPass     = "admin123"

# Бесконечный цикл опроса
while ($true) {
    # Читаем список камер из файла
    $cameras = Get-Content $baseFile | ConvertFrom-String -Delimiter ";" -PropertyNames ID, Name, System, IP
    $currentTime = Get-Date -Format "dd.MM.yyyy HH:mm:ss"
    
    # Очищаем лог-файл перед новой записью текущих статусов
    "ID;Название;Система;IP;Статус;Время_Обновления" | Out-File $logFile -Encoding UTF8
    
    foreach ($cam in $cameras) {
        $status = "FAIL" # По умолчанию камера сломана, пока не доказано обратное
        
        # --- 1. ОПРОС TRASSIR ---
        if ($cam.System -eq "Trassir") {
            try {
                # Стучимся в API Трассира за статусом каналов (порт по умолчанию 8080)
                $url = "http://$($cam.IP):8080/objects?type=channel"
                $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $trassirUser, $trassirPass)))
                $response = Invoke-RestMethod -Uri $url -Headers @{Authorization=("Basic {0}" -f $base64AuthInfo)} -TimeoutSec 3
                
                # Ищем нашу камеру в ответе и проверяем флаг активности
                $status = "OK" # Если API ответил без ошибки, сервер видит потоки
            } catch { $status = "FAIL" }
        }
        
        # --- 2. ОПРОС RVi (Dahua) ---
        elseif ($cam.System -eq "RVi") {
            try {
                # Запрос к API Dahua/RVi на предмет потери видеосигнала (Video Loss)
                $url = "http://$($cam.IP)/cgi-bin/devVideoInput.cgi?action=getVideoLoss"
                $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $rviUser, $rviPass)))
                $response = Invoke-WebRequest -Uri $url -Headers @{Authorization=("Basic {0}" -f $base64AuthInfo)} -TimeoutSec 3
                
                if ($response.Content -match "false") { $status = "OK" } else { $status = "FAIL" }
            } catch { $status = "FAIL" }
        }
        
        # --- 3. ОПРОС SpezVision ---
        else {
            # Для старого аналогового SpezVision проверяем доступность самого регистратора
            if (Test-Connection -ComputerName $cam.IP -Count 1 -Quiet) { $status = "OK" } else { $status = "FAIL" }
        }
        
        # Записываем строчку в Excel-файл
        "$($cam.ID);$($cam.Name);$($cam.System);$($cam.IP);$status;$currentTime" | Out-File $logFile -Append -Encoding UTF8
    }
    
    # Спим 5 минут (300 секунд) перед следующим кругом проверки
    Start-Sleep -Seconds 300
}
```

```cmd
$baseFile = "W:\cameras.txt"       # Список камер, который ты заполнишь
$logFile  = "W:\status_log.csv"    # Сюда скрипт будет писать живой статус

# --- ДАННЫЕ ДЛЯ ТРАССИРА И РЕГИСТРАТОРОВ ---
$trassirUser = "ТВОЙ_ЛОГИН_ТРАССИР"
$trassirPass = "ТВОЙ_ПАРОЛЬ_ТРАССИР"

# Отключаем ругань Windows на самоподписанные HTTPS-сертификаты Трассира
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
[Net.ServicePointManager]::ServerCertificateValidationCallback = {$true}

# Бесконечный цикл проверки каждые 5 минут
while ($true) {
    if (-not (Test-Path $baseFile)) {
        Write-Host "Файл со списком камер не найден на сетевом диске!"
        Start-Sleep -Seconds 10
        continue
    }

    # Читаем список наших камер из файла
    $cameras = Get-Content $baseFile | ConvertFrom-String -Delimiter ";" -PropertyNames ID, Name, System, IP
    $currentTime = Get-Date -Format "dd.MM.yyyy HH:mm:ss"
    
    # Создаем шапку таблицы в CSV файле (перезаписываем файл актуальным статусом)
    "ID;Название;Система;IP;Статус;Время_Обновления" | Out-File $logFile -Encoding UTF8
    
    foreach ($cam in $cameras) {
        $status = "FAIL" # По умолчанию камера сломана, пока не доказано обратное
        
        # 1. ЕСЛИ КАМЕРА НА СЕРВЕРАХ TRASSIR
        if ($cam.System -eq "Trassir") {
            try {
                $url = "https://$($cam.IP):8080/objects?type=channel"
                $auth = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $trassirUser, $trassirPass)))
                
                # Запрашиваем данные у сервера
                $response = Invoke-RestMethod -Uri $url -Headers @{Authorization=("Basic {0}" -f $auth)} -TimeoutSec 5
                
                # Сервер Трассира ответил успешно -> значит, сам сервер в сети
                $status = "OK" 
            } catch { 
                $status = "FAIL" 
            }
        }
        
        # 2. ЕСЛИ КАМЕРА НА СТАРЫХ РЕГИСТРАТОРАХ (RVi / SpezVision)
        else {
            # Так как старые регистраторы аналоговые, проверяем в сети ли сама "железка"
            if (Test-Connection -ComputerName $cam.IP -Count 1 -Quiet) {
                $status = "OK"
            } else {
                $status = "FAIL"
            }
        }
        
        # Записываем строчку в итоговый лог-файл для Excel
        "$($cam.ID);$($cam.Name);$($cam.System);$($cam.IP);$status;$currentTime" | Out-File $logFile -Append -Encoding UTF8
    }
    
    # Спим 5 минут (300 секунд) и идем на новый круг проверки
    Start-Sleep -Seconds 300
}
```

```cmd
$baseFile = "W:\cameras.txt"       # Твой список IP
$logFile  = "W:\status_log.csv"    # Итоговый лог для Excel

while ($true) {
    if (-not (Test-Path $baseFile)) { Start-Sleep -Seconds 10; continue }

    $cameras = Get-Content $baseFile | ConvertFrom-String -Delimiter ";" -PropertyNames ID, Name, System, IP, Port
    $currentTime = Get-Date -Format "dd.MM.yyyy HH:mm:ss"
    
    # Шапка для Excel
    "ID;Название;Система;IP;Статус;Время_Обновления" | Out-File $logFile -Encoding UTF8

        $status = "FAIL"
        
        try {
            # Проверяем, открыт ли конкретный порт видеонаблюдения на этом IP
            $connection = New-Object System.Net.Sockets.TcpClient
            $asyncResult = $connection.BeginConnect($cam.IP, [int]$cam.Port, $null, $null)
            $wait = $asyncResult.AsyncWaitHandle.WaitOne(1500, $false) # Тайм-аут 1.5 секунды
            
            if ($connection.Connected) {
                $status = "OK"
                $connection.Close()
            } else {
                $status = "FAIL"
            }
        } catch {
            $status = "FAIL"
        }

        "$($cam.ID);$($cam.Name);$($cam.System);$($cam.IP);$status;$currentTime" | Out-File $logFile -Append -Encoding UTF8
    }
    
    Start-Sleep -Seconds 300 # Проверка каждые 5 минут
}
```

```cmd
    $registrars = Get-Content $configFile | ConvertFrom-String -Delimiter ";" -PropertyNames ID, Name, System, IP, User, Pass, Channels
    $currentTime = Get-Date -Format "dd.MM.yyyy HH:mm:ss"
    
    # Записываем шапку итогового лога для Excel
    "Имя_Регистратора;Номер_Камеры;Название_Канала;Статус;Время_Обновления" | Out-File $logFile -Encoding UTF8
    
    foreach ($reg in $registrars) {
        $regChannels = [int]$reg.Channels
        
        # --- ЛОГИКА ДЛЯ RVi (Dahua API) ---
        if ($reg.System -eq "RVi") {
            try {
                # Запрашиваем у API RVi список каналов с потерей видеосигнала
                $url = "http://$($reg.IP)/cgi-bin/devVideoInput.cgi?action=getVideoLoss"
                $authInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(("{0}:{1}" -f $reg.User, $reg.Pass)))
                $response = Invoke-WebRequest -Uri $url -Headers @{Authorization=("Basic {0}" -f $authInfo)} -TimeoutSec 4
                
                # Парсим ответ API Dahua/RVi. Он обычно возвращает строки вида: channels[0].VideoLoss=false
                for ($i = 0; $i -lt $regChannels; $i++) {
                    $camNum = $i + 1
                    $searchPattern = "channels\[$i\]\.VideoLoss=(true|false)"
                    if ($response.Content -match $searchPattern) {
                        $isLoss = $Matches[1]
                        $camStatus = if ($isLoss -eq "false") { "OK" } else { "FAIL" }
                    } else {
                        $camStatus = "FAIL" # Если не смогли распарсить канал
                    }
                    # Пишем статус конкретной камеры в лог
                    "$($reg.Name);Камера $camNum;$($reg.Name)_Ch$camNum;$camStatus;$currentTime" | Out-File $logFile -Append -Encoding UTF8
                }
            } catch {
                # Если упал сам регистратор RVi — красим ВСЕ его камеры в FAIL
                for ($i = 1; $i -le $regChannels; $i++) {
                    "$($reg.Name);Камера $i;$($reg.Name)_Ch$i;FAIL;$currentTime" | Out-File $logFile -Append -Encoding UTF8
                }
            }
        }
        
        # --- ЛОГИКА ДЛЯ SpezVision (Сетевой опрос / Пинг) ---
        elseif ($reg.System -eq "SpezVision") {
            # Проверяем доступность самой «железки»
            $regOnline = Test-Connection -ComputerName $reg.IP -Count 1 -Quiet
            $status = if ($regOnline) { "OK" } else { "FAIL" }
            
            # Так как у старых SpezVision глубокий API может быть закрыт, 
            # мы транслируем статус самого регистратора на все его 16 каналов.
            # Если регистратор выключился — упадут все 16 строк в Excel.
            for ($i = 1; $i -le $regChannels; $i++) {
                "$($reg.Name);Камера $i;$($reg.Name)_Ch$i;$status;$currentTime" | Out-File $logFile -Append -Encoding UTF8
            }
        }
    }
    
    # Спим 5 минут перед следующей проверкой
    Start-Sleep -Seconds 300
}
```

```cmd
                $connection.Close()
            }
        } catch { $status = "FAIL" }
        
        # Пишем живой статус для главной панели
        "$($dev.ID);$($dev.Name);$($dev.Territory);$($dev.IP);$status;$currentTime" | Out-File $logFile -Append -Encoding UTF8
        
        # ЛОГИКА ЖУРНАЛА СБОЕВ: Если статус FAIL, фиксируем это в историю для отчета
        if ($status -eq "FAIL") {
            "$currentDate;$($dev.Name);$($dev.Territory);Отключение;$currentTime" | Out-File $historyFile -Append -Encoding UTF8
        }
    }
    
    Start-Sleep -Seconds 300 # Опрос каждые 5 минут
}
```

> Макросы (VBA) — это не просто формулы, а полноценный язык программирования Visual Basic для приложений. Они страшны для крупных компаний и их ИТ-дирекций по двум фундаментальным причинам: безопасность и совместимость.

>
> 1. 🩸 Угроза безопасности (Идеальное оружие хакеров)
>
> Макрос имеет практически безграничный доступ к операционной системе Windows от имени того пользователя, который открыл файл.
>
> Через макрос можно незаметно написать партизанский скрипт, который скачает из интернета вирус-шифровальщик, удалит файлы с компьютера или украдет пароли.
>
> Именно через документы Excel с макросами (файлы .xlsm), присланные по почте под видом «актов сверки» или «важных отчетов», происходит 80% хакерских атак на крупные предприятия.
>
> Следствие: В корпоративных версиях Windows и Office ИТ-отделы часто намертво блокируют запуск макросов на уровне групповых политик. Если ты принесешь СБ файл с макросами, при открытии у них выскочит грозное красное окно: «Запуск макросов заблокирован вашим администратором». И твоя красивая презентация сорвется.
>
> 2. 🧱 Совместимость и «ломкость» системы
>
> Макросы жестко привязаны к версии Excel, разрядности Windows (32-бит или 64-бит) и языковым пакетам.
>
> Если ты напишешь макрос на своем компьютере, а у начальника СБ будет стоять чуть более старый Office (или наоборот, новый), макрос может выдать ошибку Run-time error прямо при запуске.
>
> Экран зависнет, вылезет окно с кодом, и макрос сломается. Обычный пользователь (безопасник) испугается и закроет файл.
