```cmd
import os
import time
from PIL import Image, ImageGrab

def take_screenshot(output_path="current_state.jpg", quality=75, target_width=1280):
    """
    Делает скриншот игрового экрана, оптимизирует его для Vision LLM 
    и сохраняет в JPEG для экономии трафика.
    """
    # Снимок всего экрана (работает и в режиме "окно без рамки" для CK3)
    screenshot = ImageGrab.grab()
    
    # Пропорциональное изменение размера (LLM не нужно 4К разрешение)
    width_percent = (target_width / float(screenshot.size[0]))
    target_height = int((float(screenshot.size[1]) * float(width_percent)))
    
    optimized_image = screenshot.resize((target_width, target_height), Image.Resampling.LANCZOS)
    
    # Сохраняем в JPG сжатием
    optimized_image.convert('RGB').save(output_path, "JPEG", quality=quality)
    print(f"[*] Скриншот сохранен и оптимизирован: {output_path}")
    return output_path
```

```cmd
def process_user_voice_command(text_from_stt):
    """
    Анализирует текст от игрока. Если есть кодовое слово — добавляет скриншот.
```

```cmd
import base64

def encode_image_to_base64(image_path):
    with open(image_path, "rb") as image_file:
        return base64.b64encode(image_file.read()).decode('utf-8')

def send_to_vizier_api(api_client, model_name, command_payload, system_prompt):
    """
    Отправляет запрос советнику. Поддерживает и чистый текст, и текст + скриншот.
    """
    messages = [
        {"role": "system", "content": system_prompt}
    ]
    
    user_content = []
    # Добавляем текстовый вопрос игрока
    user_content.append({"type": "text", "text": command_payload["text"]})
    
    # Если агент прикрепил скриншот — добавляем его в тело запроса
    if command_payload["screenshot_path"]:
        base64_image = encode_image_to_base64(command_payload["screenshot_path"])
        user_content.append({
            "type": "image_url",
            "image_url": {
                "url": f"data_image/jpeg;base64,{base64_image}"
            }
        })
        
    messages.append({"role": "user", "content": user_content})
    
    # Вызов API (подходит для Polza/OpenRouter/DeepSeek/Ollama)
    response = api_client.chat.completions.create(
        model=model_name,
        messages=messages,
        max_tokens=300
    )
    
    return response.choices[0].message.content
```

>
