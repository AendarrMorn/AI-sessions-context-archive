@echo off
setlocal enabledelayedexpansion
chcp 1251 > nul

rem --- НАСТРОЙКА ПУТЕЙ ---
set "SOURCE_DIR=V:\Отдел безопасности\Фото тс сырье\Автоматизация (входящие фото)"
set "BASE_TARGET=V:\Отдел безопасности\Фото тс сырье"
set "LOG_FILE=%BASE_TARGET%\история_синхронизации.txt"

rem --- ТЕКУЩЕЕ ВРЕМЯ ДЛЯ ЗАПИСИ В ЛОГ ---
for /f "tokens=1-3 delims=. " %%a in ("%date%") do set "LOG_DATE=%%a.%%b.%%c"
set "CURRENT_TIME=%time:~0,5%"

echo Проверка наличия новых фото...
dir /b /a-d "%SOURCE_DIR%\IMG_*.jpg" >nul 2>nul
if %errorlevel% neq 0 (
    echo [%LOG_DATE% - %CURRENT_TIME%] Новых фото для поштучной сортировки не найдено. >> "%LOG_FILE%"
    goto :END
)

set "moved_count=0"
set "skipped_count=0"

rem --- ЦИКЛ ПО ВСЕМ ФОТОГРАФИЯМ ---
for %%F in ("%SOURCE_DIR%\IMG_*.jpg") do (
    set "filename=%%~nxF"
    
    rem Разрезаем имя: IMG_20260720_231608.jpg
    rem Извлекаем год, месяц, день и час
    set "f_year=!filename:~4,4!"
    set "f_month=!filename:~8,2!"
    set "f_day=!filename:~10,2!"
    set "f_hour=!filename:~13,2!"
    
    rem Убираем ведущий ноль у часа для корректной математики CMD
    if "!f_hour:~0,1!"=="0" set "f_hour=!f_hour:~1!"
    if "!f_hour!"=="" set "f_hour=0"
    
    rem Применяем правило смены: если час < 8, то это предыдущие сутки
    if !f_hour! lss 8 (
        rem Убираем ведущие нули у дня и месяца для расчетов
        if "!f_day:~0,1!"=="0" set "f_day=!f_day:~1!"
        if "!f_month:~0,1!"=="0" set "f_month=!f_month:~1!"
        
        set /a "f_day-=1"
        
        if !f_day!==0 (
            set /a "f_month-=1"
            if !f_month!==0 (
                set /a "f_year-=1"
                set "f_month=12"
            )
            set "lines=31 28 31 30 31 30 31 31 30 31 30 31"
            set /a "leap=f_year %% 4"
            if !leap!==0 set "lines=31 29 31 30 31 30 31 31 30 31 30 31"
            
            set "cnt=1"
            for %%i in (!lines!) do (
                if !cnt!==!f_month! set "f_day=%%i"
                set /a "cnt+=1"
            )
        )
        rem Возвращаем ведущие нули обратно
        if !f_day! lss 10 set "f_day=0!f_day!"
        if !f_month! lss 10 set "f_month=0!f_month!"
    )
    
    rem Получаем итоговую дату для папки (ДД.ММ.ГГГГ)
    set "TARGET_DAY=!f_day!.!f_month!.!f_year!"
    
    rem Определяем текстовое название месяца
    if "!f_month!"=="01" set "M_NAME=Январь"
    if "!f_month!"=="02" set "M_NAME=Февраль"
    if "!f_month!"=="03" set "M_NAME=Март"
    if "!f_month!"=="04" set "M_NAME=Апрель"
    if "!f_month!"=="05" set "M_NAME=Май"
    if "!f_month!"=="06" set "M_NAME=Июнь"
    if "!f_month!"=="07" set "M_NAME=Июль"
    if "!f_month!"=="08" set "M_NAME=Август"
    if "!f_month!"=="09" set "M_NAME=Сентябрь"
    if "!f_month!"=="10" set "M_NAME=Октябрь"
    if "!f_month!"=="11" set "M_NAME=Ноябрь"
    if "!f_month!"=="12" set "M_NAME=Декабрь"
    
    set "TARGET_MONTH_DIR=!M_NAME! !f_year!"
    set "FULL_TARGET_PATH=%BASE_TARGET%\!TARGET_MONTH_DIR!\!TARGET_DAY!"
    
    rem Создаем папки, если их нет
    if not exist "!FULL_TARGET_PATH!" mkdir "!FULL_TARGET_PATH!"
    
    rem Проверка на дубликат
    if exist "!FULL_TARGET_PATH!\!filename!" (
        rem Если файл уже есть в целевой папке — просто удаляем его из входящих
        del /f /q "%%F"
        set /a "skipped_count+=1"
    ) else (
        rem Если файла нет — перемещаем
        move "%%F" "!FULL_TARGET_PATH!\" > nul
        set /a "moved_count+=1"
    )
)

rem --- ЗАПИСЬ РЕЗУЛЬТАТОВ В ЛОГ ---
echo [%LOG_DATE% - %CURRENT_TIME%] Умная сортировка завершена. Перемещено новых фото: %moved_count%, удалено дубликатов: %skipped_count%. >> "%LOG_FILE%"

:END
rem --- ЗВУКОВОЙ СИГНАЛ ЗАВЕРШЕНИЯ ---
echo  
timeout /t 5 > nul
