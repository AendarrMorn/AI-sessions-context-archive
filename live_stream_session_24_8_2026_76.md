# Математическое ядро Психопроцессора (Прототип Головы Химеры)
import json

def calculate_system_state(character_name, tim, somatotype, temperament_field, individual_energy, stress_level):
    """
    Рассчитывает натяжение Дроби Реальности, силу инстинкта и аспектный перегрев.
    """
    # 1. Ограничения Сомы (Тело как Воля)
    energy_burn_rate = 2.5 if somatotype == "Эктоморф" else 1.2  # Эктоморфы жгут энергию быстрее
    is_buffer_low = individual_energy < 30
    
    # 2. Бунт Косного Бухгалтера (Инстинкт самосохранения)
    accountant_power = 0.9 if is_buffer_low else 0.3
    if temperament_field in ["ХФ", "МФ"]:  # Инертные поля дают бухгалтеру больше веса
        accountant_power += 0.1
        
    # 3. Ментальное давление ТИМа (Тело как Представление)
    # Определяем силу ментального прорыва вопреки телу (Пассионарный потенциал ТИМа в стрессе)
    if tim in ["ИЛЭ (Дон-Кихот)", "ЭИЭ (Гамлет)"]:
        willpower_override_potential = 0.8  # Высокий идейный/эмоциональный прорыв
    else:
        willpower_override_potential = 0.2  # Дюма (СЭИ) слушает тело и гомеостаз
        
    # 4. Расчет Аспектного Перегрева
    # Если стресс высокий, а потенциал прорыва давит, ментал начинает выжигать витал
    aspect_overheat = 0.0
    if stress_level > 50:
        aspect_overheat = (stress_level * willpower_override_potential) / (individual_energy / 100.0)
    
    # Ограничиваем перегрев шкалой до 100%
    aspect_overheat = min(100.0, round(aspect_overheat, 1))
    
    # 5. Системный вердикт (Девиация относительно Нормы)
    if aspect_overheat > 75:
        deviation = "Критический аспектный перегрев (Ментальная тирания выжигает сому)"
    elif accountant_power > 0.8 and willpower_override_potential < 0.5:
        deviation = "Бунт Бухгалтера (Энергетический анабиоз, ментал подавлен ленью/страхом)"
    else:
        deviation = "Идеальная Норма (Динамическое натяжение вертикали стабильно)"
        
    # Формируем JSON-паспорт для отправки в LLM-Архитектор
    passport = {
        "Character": character_name,
        "Vertical_Stack": {
            "Layer_4_Representation_TIM": tim,
            "Layer_2_3_Psychophysiology": {
                "Field": temperament_field,
                "Stress_Index": stress_level,
                "Aspect_Overheat_Rate": f"{aspect_overheat}%"
            },
            "Layer_1_Body_Per_Se": {
                "Somatotype": somatotype,
                "Energy_Reserves": f"{individual_energy}%",
                "Accountant_Lockout_Power": round(accountant_power, 2)
            }
        },
        "System_Verdict": deviation
    }
    
    return passport

# Тестируем симуляцию "Болото" для нашей троицы при падении сил и высоком стрессе
swamp_simulation = [
    calculate_system_state("Дон Кихот", "ИЛЭ (Дон-Кихот)", "Эктоморф", "ХС", individual_energy=25, stress_level=80),
    calculate_system_state("Гамлет", "ЭИЭ (Гамлет)", "Эктоморф", "МС", individual_energy=20, stress_level=95),
    calculate_system_state("Дюма", "СЭИ (Дюма)", "Эндоморф", "МФ", individual_energy=28, stress_level=60)
]

print(json.dumps(swamp_simulation, indent=2, ensure_ascii=False))

import math

class ReservoirM2:
    def __init__(self, amplitude, frequency, baseline_shift):
        self.A = amplitude         # Амплитуда
        self.B = frequency         # Частота флуктуаций
        self.C = baseline_shift    # Базовый уровень емкости (ось Y)
        self.time = 0.0
        
    def get_current_value(self, dt, external_drain=0):
        """Рассчитывает текущее значение бака с учетом синусоиды и внешнего истощения"""
        self.time += dt
        # Естественное дыхание бака минус внешнее давление среды (например, болота)
        current_capacity = self.A * math.sin(self.B * self.time) + self.C - external_drain
        return max(0.0, current_capacity)

    def evolve_reservoir(self, delta_amplitude, delta_shift):
        """Протокол прокачки персонажа: расширяем амплитуду и сдвигаем ось Y вверх"""
        self.A += delta_amplitude
        self.C += delta_shift

> Вспомни наш разговор про синусоиды параметров (М1–М4). Откуда игре брать глобальные космические циклы, которые меняют емкость баков всех NPC в деревне? Из этого самого календаря!
>
> Пентаграмматон Старшего Круга (Пта, Сатх, Танис, Тэт, Маат) — 5-ричный цикл:
>
> Это цикл Внутренней Статики и Онтологии.
>
> Он задает фундаментальную метрику законов природы в Изумрудной Гавани. Например, в день Пта (Творца) — Анаторы горят ярче, онтологическая стабильность удерживается легче. В день Сатха — Радужный Туман Хенналана становится агрессивнее, аспектный перегрев у интуитов нарастает лавинообразно.

> Септенер Младшего Круга (Нейт, Нептис, Таэла, Хермис, Майя, Белис, Амента) — 7-ричный цикл:

> Это цикл Витальной Динамики и Психофизиологии.

> 🌲 Визуальное Дерево Состояний (State Tree) в Unreal Engine 5
>
> Чтобы ты видел живой результат и работа оставалась игрой, давай зафиксируем, как мы соберем визуальное Дерево Состояний для андроидов в UE5, учитывая этот календарь.
>
> Unreal Engine 5 имеет встроенный мощный инструмент — State Tree. Это граф, состоящий из кубиков-состояний. Каждая ветка активируется только при выполнении математических условий в Blackboard.
>
> Схема State Tree андроида в нашей песочнице:
>
> Root (Корень)
>
> 🎄 Ветка 1: [Emergency_Override] (Слой 0 — Ипостасный узел / Willpower Override).
>
> Условие: Ontological_Stability < 20% или включен протокол «Пассионарный Долг».
>
> Действие: Игнорировать все баки тела per se. Включить анимацию «Безумный Пророк». Выдать ТИМную речь Инквизитора Гамлета.
>
> 🍖 Ветка 2: [Vital_Urgent] (Бунт Бухгалтера — Уровень 1).
>
> Условие: Energy_M2 < 15% и Боги Младшего Круга в фазе Аменты (Лень/Анабиоз).
>
> Действие: Перехватить управление у ментальных задач. Послать команду движку: MoveTo(Tavern) -> PlayAnimation(Eating).
>
> 💼 Ветка 3: [TIM_Social_Behavior] (Идеальная Норма — Уровень 4).
>
> Условие: Баки в норме, Анатор горит.
>
> Действие: ТИМ Штирлиц идет чинить частокол (Task_Repair). ТИМ Дон Кихот идет искать скрытые смыслы на поле (Task_Explore).
>
> Ты сидишь перед экраном UE5. На календаре игры выпадает сочетание [Сатх + Майя] (Высокий ментальный перегрев + Лабильность чувств).
>
> Ты видишь, как андроид-Гамлет у частокола мгновенно срывается с ветки 3 (Социальное поведение) в ветку 1 (Экзистенциальный override), потому что Сатх пробил его онтологическую стабильность! Он начинает метаться по сцене, анимации сменяют друг друга, а над головой всплывает сгенерированный нашей LLM текст. Игра началась.
