@echo off
setlocal enabledelayedexpansion
chcp 1251 > nul

rem --- НАСТРОЙКА ПУТЕЙ ---
set "SOURCE_DIR=V:\Отдел безопасности\Фото тс сырье\Автоматизация (входящие фото)"
set "BASE_TARGET=V:\Отдел безопасности\Фото тс сырье"

echo Проверка наличия новых фото...
dir /b /a-d "%SOURCE_DIR%" >nul 2>nul
if %errorlevel% neq 0 (
    echo Фотографий для сортировки не найдено. Работу завершаю.
    goto :END
)

rem --- ВЫЧИСЛЕНИЕ ВЧЕРАШНЕЙ ДАТЫ ---
for /f "tokens=1-3 delims=. " %%a in ("%date%") do (
    set "d=%%a"
    set "m=%%b"
    set "y=%%c"
)

rem Убираем ведущие нули для правильного математического расчета в CMD
if "!d:~0,1!"=="0" set "d=!d:~1!"
if "!m:~0,1!"=="0" set "m=!m:~1!"

set /a "d-=1"

if %d%==0 (
    set /a "m-=1"
    if !m!==0 (
        set /a "y-=1"
        set "m=12"
    )
    set "lines=31 28 31 30 31 30 31 31 30 31 30 31"
    set /a "leap=y %% 4"
    if !leap!==0 set "lines=31 29 31 30 31 30 31 31 30 31 30 31"
    
    set "count=1"
    for %%i in (!lines!) do (
        if !count!==!m! set "d=%%i"
        set /a "count+=1"
    )
)

if %d% lss 10 set "d=0%d%"
if %m% lss 10 set "m=0%m%"
set "YESTERDAY=%d%.%m%.%y%"

rem --- ОПРЕДЕЛЕНИЕ НАЗВАНИЯ МЕСЯЦА НА РУССКОМ ---
if "%m%"=="01" set "MONTH_NAME=Январь"
if "%m%"=="02" set "MONTH_NAME=Февраль"
if "%m%"=="03" set "MONTH_NAME=Март"
if "%m%"=="04" set "MONTH_NAME=Апрель"
if "%m%"=="05" set "MONTH_NAME=Май"
if "%m%"=="06" set "MONTH_NAME=Июнь"
if "%m%"=="07" set "MONTH_NAME=Июль"
if "%m%"=="08" set "MONTH_NAME=Август"
if "%m%"=="09" set "MONTH_NAME=Сентябрь"
if "%m%"=="10" set "MONTH_NAME=Октябрь"
if "%m%"=="11" set "MONTH_NAME=Ноябрь"
if "%m%"=="12" set "MONTH_NAME=Декабрь"

set "FOLDER_MONTH=%MONTH_NAME% %y%"
set "FINAL_TARGET_DIR=%BASE_TARGET%\%FOLDER_MONTH%\%YESTERDAY%"

echo.
echo Вчерашний день определен как: %YESTERDAY%
echo Целевая папка: %FINAL_TARGET_DIR%
echo.

rem --- АВТОМАТИЧЕСКОЕ СОЗДАНИЕ СТРУКТУРЫ ПАПОК ---
if not exist "%FINAL_TARGET_DIR%" (
    echo Создаю дерево папок: %FOLDER_MONTH% -> %YESTERDAY%
    mkdir "%FINAL_TARGET_DIR%"
)

rem --- ПЕРЕМЕЩЕНИЕ ФАЙЛОВ ---
echo Перенос фотографий...
move "%SOURCE_DIR%\*.*" "%FINAL_TARGET_DIR%\"

echo.
echo Все фотографии успешно перемещены!

:END
echo.
timeout /t 7

// @name         Синхронизатор сессий ИИ в GitHub

// @match        *://*/* // Скрипт будет искать интерфейс чата на открытой странице

(function() {
    'use strict';

    // =======================================================
    //                НАСТРОЙКА КЛЮЧЕЙ И РЕПОЗИТОРИЯ
    // =======================================================
    const GITHUB_USER = 'ВАШ_ЛОГИН_НА_GITHUB';
    const GITHUB_REPO = 'НАЗВАНИЕ_РЕПОЗИТОРИЯ';
    const GITHUB_TOKEN = 'СЮДА_ВСТАВЬТЕ_ТОКЕН_GITHUB_REPO_CLASSIC';
    // =======================================================

    // Функция для сбора диалога со страницы чата
    function collectChatData() {
        let markdownText = '# История совместной сессии\n\n';
        
        // Находим все блоки сообщений на странице. 
        // Примечание: Селекторы подстраиваются под структуру элементов разметки чата.
        const msgBlocks = document.querySelectorAll('div[data-message-author-role], .message-content, article');
        
        if (msgBlocks.length === 0) {
            alert('Блоки сообщений не найдены. Возможно, нужно скорректировать селекторы разметки страницы.');
            return null;
        }

        msgBlocks.forEach((block) => {
            const isUser = block.innerText.includes('Вы') || block.getAttribute('data-message-author-role') === 'user';
            const text = block.innerText.replace(/Копировать|Поделиться|Оценить|Экспортировать/g, '').trim();
            
            if (text.length > 0) {
                if (isUser) {
                    markdownText += `### 👤 Пользователь:\n${text}\n\n`;
                } else {
                    markdownText += `### 🤖 Ассистент:\n${text}\n\n---\n\n`;
                }
            }
        });

        return markdownText;
    }

    // Функция отправки через API GitHub
    function pushToGitHub(content) {
        const date = new Date();
        const timestamp = `${date.getDate()}_${date.getMonth()+1}_${date.getFullYear()}_${date.getHours()}${date.getMinutes()}`;
        const fileName = `session_${timestamp}.md`;
        
        // Переводим текст в Base64 для передачи через API
        const base64Content = btoa(unescape(encodeURIComponent(content)));

        const url = `https://github.com{GITHUB_USER}/${GITHUB_REPO}/contents/contexts/${fileName}`;
        
        GM_xmlhttpRequest({
            method: 'PUT',
            url: url,
            headers: {
                'Authorization': `token ${GITHUB_TOKEN}`,
                'Content-Type': 'application/json',
                'User-Agent': 'Tampermonkey-AI-Bot'
            },
            data: JSON.stringify({
                message: `Автоматическое сохранение сессии ${timestamp}`,
                content: base64Content
            }),
            onload: function(response) {
                if (response.status === 201 || response.status === 200) {
                    alert(`УСПЕШНО! Сессия сохранена в GitHub как ${fileName}`);
                } else {
                    alert('Ошибка GitHub: ' + response.responseText);
                }
            },
            onerror: function(error) {
                alert('Сетевая ошибка при отправке в GitHub.');
            }
        });
    }

    // Создаем красивую кнопку интерфейса
    function createExportButton() {
        const btn = document.createElement('button');
        btn.innerHTML = '📂 Отправить сессию в GitHub';
        btn.style.position = 'fixed';
        btn.style.top = '10px';
        btn.style.right = '50px';

        btn.style.cursor = 'pointer';
        btn.style.fontWeight = 'bold';
        btn.style.fontFamily = 'Segoe UI, sans-serif';

        btn.addEventListener('click', () => {
            const chatContent = collectChatData();
            if (chatContent) {
                pushToGitHub(chatContent);
            }
        });

        document.body.appendChild(btn);
    }

    // Запускаем кнопку после загрузки страницы
    setTimeout(createExportButton, 3000);
})();

        let markdownText = '# История совместной сессии\n\n';
        let hasContent = false;
        
        // Ищем все элементы, похожие на сообщения (вопросы и ответы)
        const messages = document.querySelectorAll('div[data-message-author-role], .message-content, article, .chat-message');
        
        messages.forEach((msg) => {
            let text = msg.innerText;
            // Чистим текст от интерфейсного мусора и кнопок под сообщениями
            text = text.replace(/Копировать|Поделиться|Оценить|Экспортировать|Хорошо|Плохо/g, '').trim();
